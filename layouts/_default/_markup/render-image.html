{{- $class := "wide" -}} {{- $width := "" -}} {{- $caption := "" -}} {{- $credit := "" -}} {{- $title := .Title -}} {{/* Parse the title attribute. Supported forms: "Some caption" → wide + caption "width=600" → wide + constrain width "class=inline" → stay in the prose column "class=full" → full-bleed "class=wide; caption; credit=Photo: YM" → combinable via ";" Default class is "wide" if no class= is set. */}} {{- if $title -}} {{- $parts := split $title ";" -}} {{- range $parts -}} {{- $p := trim . " " -}} {{- if hasPrefix $p "width=" -}} {{- $width = strings.TrimPrefix "width=" $p -}} {{- else if hasPrefix $p "class=" -}} {{- $class = strings.TrimPrefix "class=" $p -}} {{- else if hasPrefix $p "credit=" -}} {{- $credit = strings.TrimPrefix "credit=" $p -}} {{- else if $p -}} {{- $caption = $p -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $caption $credit -}} {{- end -}}
